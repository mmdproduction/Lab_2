cmake_minimum_required(VERSION 3.16)
project(Lab_2 CXX)
set(CMAKE_CXX_STANDARD 20)

# Опции для управления сборкой тестов и GUI
option(BUILD_LAB2_TESTS "Build tests for Lab_2" OFF)
option(BUILD_LAB2_GUI "Build GUI for Lab_2" OFF)

set(SRC 
    ${CMAKE_CURRENT_SOURCE_DIR}/src/)
set(GUI 
    ${CMAKE_CURRENT_SOURCE_DIR}/gui/)

set(HEADER_DIR
    ${SRC}DynamicArray/
    ${SRC}Exception/
    ${SRC}IEnumerator/
    ${SRC}LinkedList/
    ${SRC}Sequence/
    ${SRC}Sequence/LazySequence/
    ${SRC}IEnumerator/LazyEnumerators/
    ${SRC}Test/
)

set(HEADER_FILES
    ${SRC}DynamicArray/DynamicArray.hpp
    ${SRC}Exception/Exception.hpp
    ${SRC}IEnumerator/IEnumerator.hpp
    ${SRC}IEnumerator/Iterator.hpp
    ${SRC}LinkedList/LinkedList.hpp
    ${SRC}Sequence/Sequence.hpp
    ${SRC}Sequence/BitSequence.hpp
    ${SRC}IEnumerator/LazyEnumerators/LazyEnumerators.hpp
    ${SRC}Sequence/LazySequence/SequenceView.hpp
    ${SRC}Test/TestRunner.hpp
)

set(TEMPLATE_FILES
    ${SRC}DynamicArray/DynamicArray.tpp
    ${SRC}IEnumerator/IEnumerator.tpp
    ${SRC}IEnumerator/Iterator.tpp
    ${SRC}LinkedList/LinkedList.tpp
    ${SRC}Sequence/Sequence.tpp
    ${SRC}Sequence/ListSequence.tpp
    ${SRC}Sequence/ArraySequence.tpp
    ${SRC}IEnumerator/LazyEnumerators/IndexEnumerator.tpp
    ${SRC}IEnumerator/LazyEnumerators/ConcatEnumerator.tpp
    ${SRC}IEnumerator/LazyEnumerators/SkipEnumerator.tpp
    ${SRC}IEnumerator/LazyEnumerators/MapEnumerator.tpp
    ${SRC}IEnumerator/LazyEnumerators/FilterEnumerator.tpp
    ${SRC}Sequence/LazySequence/MapSequenceView.tpp
    ${SRC}Sequence/LazySequence/FilterSequenceView.tpp
    ${SRC}Sequence/LazySequence/ConcatSequenceView.tpp
    ${SRC}Sequence/LazySequence/SubSequenceView.tpp
)

set(SOURCE_FILES
    ${SRC}Sequence/BitSequence.cpp
    ${SRC}Test/TestRunner.cpp
)


add_library(${PROJECT_NAME} STATIC
    ${SOURCE_FILES}
    ${HEADER_FILES}
)

target_include_directories(${PROJECT_NAME} PUBLIC ${HEADER_DIR})

if(BUILD_LAB2_TESTS OR BUILD_LAB2_GUI)

    if(NOT TARGET gtest AND NOT TARGET gtest_main)
        add_subdirectory(libs/gtest)
    endif()
    
    if(BUILD_LAB2_GUI AND NOT TARGET OpenGUI)
        if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/libs/OpenGUI/CMakeLists.txt)
            add_subdirectory(libs/OpenGUI)
        else()
            message(WARNING "OpenGUI not found, GUI disabled")
        endif()
    endif()
endif()

if(BUILD_LAB2_TESTS)
    target_link_libraries(${PROJECT_NAME} PUBLIC gtest)
endif()


if(BUILD_LAB2_GUI AND TARGET OpenGUI)
    add_executable(lab2_gui
        ${GUI}main.cpp
    )
    
    target_link_libraries(lab2_gui PUBLIC
        ${PROJECT_NAME}
        OpenGUI
    )
    
    target_include_directories(lab2_gui PUBLIC
        ${GUI}
        ${GUI}View/
        ${GUI}Controller/
        ${GUI}Model/
    )
endif()


if(BUILD_LAB2_TESTS)
    enable_testing()
    
    set(TEST_SOURCE_FILES 
        tests/DynamicArrayTest.cpp
        tests/LinkedListTest.cpp
        tests/ArraySequenceTest.cpp
        tests/ListSequenceTest.cpp
    )
    
    add_executable(lab2_tests
        ${TEST_SOURCE_FILES}
    )
    
    target_link_libraries(lab2_tests PRIVATE 
        ${PROJECT_NAME} 
        gtest_main
    )
    
    target_include_directories(lab2_tests PRIVATE ${HEADER_DIR})
    
    include(GoogleTest)
    gtest_discover_tests(lab2_tests)
endif()
